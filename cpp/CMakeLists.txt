cmake_minimum_required(VERSION 3.20)
project(ma2a_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_package(Threads REQUIRED)
add_library(ma2a_core INTERFACE)
target_include_directories(ma2a_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(ma2a_core INTERFACE OpenSSL::Crypto)
set(RESOLUTIVE_ROUTING_SOURCE_DIR "" CACHE PATH "Path to resolutive-routing checkout")
add_executable(ma2a_assertions_enabled_test tests/assertions_enabled_test.cpp)
target_link_libraries(ma2a_assertions_enabled_test PRIVATE ma2a_core)
add_executable(ma2a_core_contract_test tests/contract_test.cpp)
target_link_libraries(ma2a_core_contract_test PRIVATE ma2a_core)
add_executable(ma2a_job_framing_test tests/job_framing_test.cpp)
target_link_libraries(ma2a_job_framing_test PRIVATE ma2a_core)
add_executable(ma2a_python_interop_test tests/python_interop_test.cpp)
target_link_libraries(ma2a_python_interop_test PRIVATE ma2a_core)
add_executable(ma2a_wire_contract_test tests/wire_contract_test.cpp)
target_link_libraries(ma2a_wire_contract_test PRIVATE ma2a_core)
add_executable(ma2a_tcp_crosslang_client tests/tcp_crosslang_client.cpp)
target_link_libraries(ma2a_tcp_crosslang_client PRIVATE ma2a_core)
add_executable(ma2a_tcp_crosslang_signed_client tests/tcp_crosslang_signed_client.cpp)
target_link_libraries(ma2a_tcp_crosslang_signed_client PRIVATE ma2a_core)
add_executable(ma2a_failure_notice_test tests/failure_notice_test.cpp)
target_link_libraries(ma2a_failure_notice_test PRIVATE ma2a_core)
add_executable(ma2a_execution_orchestrator_test tests/execution_orchestrator_test.cpp)
target_link_libraries(ma2a_execution_orchestrator_test PRIVATE ma2a_core)
add_executable(ma2a_resilient_execution_test tests/resilient_execution_test.cpp)
target_link_libraries(ma2a_resilient_execution_test PRIVATE ma2a_core)
add_executable(ma2a_authenticated_tcp_adversarial_test tests/authenticated_tcp_adversarial_test.cpp)
target_link_libraries(ma2a_authenticated_tcp_adversarial_test PRIVATE ma2a_core Threads::Threads)
if(RESOLUTIVE_ROUTING_SOURCE_DIR)
 add_executable(ma2a_routing_failover_driver tests/routing_failover_driver.cpp)
 target_include_directories(ma2a_routing_failover_driver PRIVATE ${RESOLUTIVE_ROUTING_SOURCE_DIR}/cpp/include)
 add_executable(ma2a_real_routing_execution_test tests/real_routing_execution_test.cpp)
 target_link_libraries(ma2a_real_routing_execution_test PRIVATE ma2a_core)
 target_include_directories(ma2a_real_routing_execution_test PRIVATE ${RESOLUTIVE_ROUTING_SOURCE_DIR}/cpp/include)
 add_executable(ma2a_authenticated_tcp_resilient_execution_test tests/authenticated_tcp_resilient_execution_test.cpp)
 target_link_libraries(ma2a_authenticated_tcp_resilient_execution_test PRIVATE ma2a_core Threads::Threads)
 target_include_directories(ma2a_authenticated_tcp_resilient_execution_test PRIVATE ${RESOLUTIVE_ROUTING_SOURCE_DIR}/cpp/include)
endif()

set(MA2A_ASSERT_TEST_TARGETS
 ma2a_assertions_enabled_test
 ma2a_core_contract_test
 ma2a_job_framing_test
 ma2a_python_interop_test
 ma2a_wire_contract_test
 ma2a_failure_notice_test
 ma2a_execution_orchestrator_test
 ma2a_resilient_execution_test
 ma2a_authenticated_tcp_adversarial_test
 ma2a_tcp_crosslang_client
 ma2a_tcp_crosslang_signed_client
 ma2a_real_routing_execution_test
 ma2a_authenticated_tcp_resilient_execution_test
)
foreach(target IN LISTS MA2A_ASSERT_TEST_TARGETS)
 if(TARGET ${target})
  if(MSVC)
   target_compile_options(${target} PRIVATE /UNDEBUG)
  else()
   target_compile_options(${target} PRIVATE -UNDEBUG)
  endif()
 endif()
endforeach()
